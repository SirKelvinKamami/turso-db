Quick Start

Integrate TursoDB into your projects

REST API

Your API base URL:

https://turso-db-8svn.onrender.com

List databases:

curl -H "Authorization: Bearer YOUR_TOKEN" \
https://turso-db-8svn.onrender.com/v1/databases
JavaScript / Node.js

Fetch wrapper:

const API = "https://turso-db-8svn.onrender.com/v1";
const TOKEN = "YOUR_TOKEN";

async function query(db, sql) {
const r = await fetch(`${API}/databases/${db}/query`, {
method: "POST",
headers: { "Authorization": `Bearer ${TOKEN}`, "Content-Type": "application/json" },
body: JSON.stringify({ sql })
});
return r.json();
}
Python

Requests library example:

import requests

API = "https://turso-db-8svn.onrender.com/v1"
HEADERS = {"Authorization": "Bearer YOUR_TOKEN"}

def query(db, sql):
r = requests.post(f"{API}/databases/{db}/query",
json={"sql": sql}, headers=HEADERS)
return r.json()
Go

Simple HTTP client:

package main

import ("fmt"; "net/http"; "io")

func main() {
url := "https://turso-db-8svn.onrender.com/v1/health"
resp, _ := http.Get(url)
body, _ := io.ReadAll(resp.Body)
fmt.Println(string(body))
}
